% Shared manuscript review markup. Load once in the document preamble.
% After other package declarations, use:
% \input{review-macro.tex}
\RequirePackage{amsmath}
\RequirePackage{xcolor}
\RequirePackage[normalem]{ulem}

% Additions work in text and math mode; unchanged content needs no wrapper.
\newcommand{\reviewadd}[1]{{\color{blue}#1}}

% Text deletions. Protect inline math/references with \mbox when necessary.
\newcommand{\reviewdel}[1]{{\color{red}\sout{#1}}}

% Math deletions: use inside math mode, one expression or alignment cell at a time.
% Match the current math style and separate an adjacent replacement slightly.
\newcommand{\reviewdelmath}[1]{\mathchoice
  {\text{\color{red}\sout{$\displaystyle #1$}}}
  {\text{\color{red}\sout{$\textstyle #1$}}}
  {\text{\color{red}\sout{$\scriptstyle #1$}}}
  {\text{\color{red}\sout{$\scriptscriptstyle #1$}}}\,}
